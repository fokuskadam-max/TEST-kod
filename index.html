// Made with ♥ by Ducky // https://christieman.com/ // - I noticed buggy scaling on mouseout for firefox, if you know why I'd love to hear it! // Degree of Rotation (try .2, .2) or higher for some fun const xRatio = 0.1; const yRatio = 0.15; const posters = document.querySelectorAll('.poster'); const onHover = (poster, shine, { offsetX, offsetY }) => { const width = poster.clientWidth; const height = poster.clientHeight; const dx = offsetX - (width / 2); const dy = offsetY - (height / 2); shine.style.transform = `translate(${dx}px, ${dy}px)`; poster.style.transform = `rotateY(${-dx * yRatio}deg) rotateX(${dy * xRatio}deg)`; }; const reset = (poster) => { poster.style.transform = ``; }; posters.forEach(poster => { const image = poster.querySelector('.poster__image'); const shine = poster.querySelector('.poster__shine'); image.addEventListener('mousemove', e => onHover(image, shine, e)); image.addEventListener('mouseout', () => reset(image)); });
